      private static void calcLength(Object[] objList) {
        for (int i = 0; i < objList.length; i++) {
            if (objList[i] instanceof Measurable) {
                try {
                    System.out.println("Длина объекта ( " + (Object) objList[i].getClass().getName() + " ):" + ((Measurable) objList[i]).getLength());
                } catch (Exception e) {
                    System.out.println("Ошибка при вычислении длины объекта: " + e.getMessage());
                }
            } else {
                System.out.println("Интерфейс нереализован");
            }
        }
    }

    public static void main(String[] args) {
        MyPoint[] pointList = {new MyPoint(1, 5),
                new MyPoint(2, 8),
                new MyPoint(5, 3),
                new MyPoint(8, 9)};

        MyLine line = new MyLine(pointList[0], pointList[2]);
        MyPolyLine polyLine = new MyPolyLine(pointList);
        MyPolyLineClose plc = new MyPolyLineClose(pointList);
